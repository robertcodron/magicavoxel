// xs ripple [Height] [Wavelength]
// Concentric ripple whose phase follows the animation, so that the generated frames
// loop: xs -n 12 -cur ripple
//
// xs_begin
// author : 'magicavoxel-rs'
// arg : { name = 'Height'      var = 'm_height'      range = '1 64'  value = '6'  step = '1'  precision = '0' }
// arg : { name = 'Wavelength'  var = 'm_wavelength'  range = '2 64'  value = '12' step = '1'  precision = '0' }
// xs_end

const float TAU = 6.28318530718;

float map(vec3 v) {
	vec2 c = i_volume_size.xy * 0.5;
	float r = length(v.xy - c);
	float phase = TAU * iFrame / max(iNumFrames, 1.0);
	float h = i_volume_size.z * 0.5 + m_height * sin(r / m_wavelength * TAU - phase);
	return v.z < h ? i_color_index : 0.0;
}
